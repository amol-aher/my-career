Code Example — Service Object Pattern (Rails)

# app/services/create_order.rb
class CreateOrder
  Result = Struct.new(:success?, :order, :errors)

  def initialize(user:, params:)
    @user = user
    @params = params
  end

  def call
    order = nil
    ActiveRecord::Base.transaction do
      order = Order.create!(user: @user, status: 'pending')
      params[:items].each do |item|
        order.line_items.create!(product_id: item[:product_id], quantity: item[:qty])
      end
    end
    Result.new(true, order, nil)
  rescue ActiveRecord::RecordInvalid => e
    Result.new(false, nil, e.record.errors.full_messages)
  end

  private
  attr_reader :user, :params
end
